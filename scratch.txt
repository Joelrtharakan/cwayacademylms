  const { programId } = req.params;
  const studentId = req.user!.id;

  const enrollment = await prisma.programEnrollment.findUnique({
    where: { studentId_programId: { studentId, programId } },
    include: {
      program: {
        include: {
          courses: {
            include: {
              enrollments: { where: { studentId } }
            }
          }
        }
      }
    }
  });
